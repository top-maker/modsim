    def require_bindable(host: str, port: Any, kind: int = socket.SOCK_STREAM) -> None:
    """Fail early with a clear message when binding needs elevated rights."""
    with socket.socket(socket.AF_INET, kind) as probe:
        try:
            probe.bind((host, int(port)))
        except PermissionError as exc:
            raise OSError(privileged_port_hint(port)) from exc
        except OSError:
            # Occupied ports and other failures are reported by the server itself.
            return

    async def run_async_server(args: Any) -> Any:
    """Run server."""
    txt = f"Start ASYNC server, listening on {args.port} - {args.comm}"
    _logger.info(txt)
    server: Any
    if args.comm == "tcp":
        address: tuple[str, int] = (
            getattr(args, "bind", None)
            or ("127.0.0.1" if getattr(args, "runtimes", None) else ""),
            args.port,
        )
        require_bindable(address[0], args.port)
        server = ModbusTcpServer(
            context=args.context,
            identity=args.identity,
            address=address,
            framer=args.framer,
        )
        if hasattr(args, "runtimes"):
            from .runtime import RuntimeContext

            server.context = RuntimeContext(args.runtimes)
        tasks = []
        try:
            if not await server.listen():
                raise OSError(
                    "Could not start Modbus TCP listener on port %s" % args.port
                )
            tasks = [
                asyncio.create_task(device.run())
                for device in getattr(args, "runtimes", [])
            ]
            await asyncio.gather(server.serving, *tasks)
        finally:
            for task in tasks:
                task.cancel()
            await asyncio.gather(*tasks, return_exceptions=True)
            await server.shutdown()
        return server
    elif args.comm == "udp":
        address = ("127.0.0.1", args.port)
        require_bindable(address[0], args.port, socket.SOCK_DGRAM)
        server = ModbusUdpServer(
            context=args.context,
            identity=args.identity,
            address=address,
            framer=args.framer,
        )
        try:
            if not await server.listen():
                raise OSError(
                    "Could not start Modbus UDP listener on port %s" % args.port
                )
            await server.serving
        finally:
            await server.shutdown()
        return server
    elif args.comm == "serial":
        await StartAsyncSerialServer(
            context=args.context,
            identity=args.identity,
            port=args.port,
            framer=args.framer,
        )
        return None
    elif args.comm == "tls":
        address = ("", args.port)
        require_bindable(address[0], args.port)
        # Create SSL context from cert and key files
        sslctx = ssl.create_default_context(ssl.Purpose.CLIENT_AUTH)
        try:
            sslctx.load_cert_chain(
                certfile="certificates/pymodbus.crt",
                keyfile="certificates/pymodbus.key",
            )
        except OSError as exc:
            raise OSError(
                "TLS needs certificates/pymodbus.crt and certificates/pymodbus.key "
                f"in the working directory: {exc}"
            ) from exc

        server = ModbusTlsServer(
            context=args.context,
            identity=args.identity,
            address=address,
            framer=args.framer,
            sslctx=sslctx,
        )
        try:
            if not await server.listen():
                raise OSError(
                    "Could not start Modbus TLS listener on port %s" % args.port
                )
            await server.serving
        finally:
            await server.shutdown()
        return server
    raise ValueError(f"Unknown communication type: {args.comm}")

    def setup_server(args: Any) -> Any:
    """Run server setup."""
    # The datastores only respond to the addresses that are initialized
    # If you initialize a DataBlock to addresses of 0x00 to 0xFF, a request to
    # 0x100 will respond with an invalid address exception.
    # This is because many devices exhibit this kind of behavior (but not all)
    if getattr(args, "config", None):
        from .deployment import configure_devices

        args.context = configure_devices(args)
        _logger.info(
            "Model initialization: %s",
            json.dumps(args.model_initialization, sort_keys=True),
        )
    if not args.context:
        _logger.info("Create datastore")
        # build your payload
        co_builder = BinaryPayloadBuilder(byteorder=Endian.BIG, wordorder=Endian.BIG)
        # address=0, bytes=3
        for _ in range(3):
            # The bits within each byte are big-endian
            # The coils status (from the first position) = [1,0,1,0,0,0,0,1]
            co_builder.add_bits([True, False, True, False, False, False, False, True])

        hr_builder = BinaryPayloadBuilder(byteorder=Endian.BIG, wordorder=Endian.BIG)
        # address=0, bytes=8
        hr_builder.add_16bit_uint(1)
        hr_builder.add_16bit_uint(2)
        hr_builder.add_16bit_uint(3)
        hr_builder.add_16bit_uint(4)
        # address=8, bytes=8
        hr_builder.add_16bit_int(-1)
        hr_builder.add_16bit_int(-2)
        hr_builder.add_16bit_int(-3)
        hr_builder.add_16bit_int(-4)
        # address=16, bytes=8
        hr_builder.add_32bit_uint(12345678)
        hr_builder.add_32bit_uint(12345678)
        # address=24, bytes=8
        hr_builder.add_32bit_int(-12345678)
        hr_builder.add_32bit_int(-12345678)
        # address=32, bytes=8
        hr_builder.add_32bit_float(12.34)
        hr_builder.add_32bit_float(-12.34)
        # address=40, bytes=8
        hr_builder.add_64bit_uint(12345678900)
        # address=48, bytes=8
        hr_builder.add_64bit_int(-12345678900)
        # address=56, bytes=16
        hr_builder.add_64bit_float(123.45)
        hr_builder.add_64bit_float(-123.45)
        # address=72, bytes=?
        hr_builder.add_string("this is a string")

        # use that payload in the data store
        co_values = co_builder.to_coils()
        hr_values = hr_builder.to_registers()
        args.context = SimDevice(
            id=0,
            simdata=(
                [SimData(0, values=co_values, datatype=DataType.BITS)],
                [SimData(10000, values=co_values, datatype=DataType.BITS)],
                [SimData(40000, values=hr_values, datatype=DataType.REGISTERS)],
                [SimData(30000, values=hr_values, datatype=DataType.REGISTERS)],
            ),
        )

    # initialize the server information
    # If you don't set this or any fields, they are defaulted to empty strings.
    args.identity = ModbusDeviceIdentification(
        info_name={
            "VendorName": "Pymodbus",
            "ProductCode": "PM",
            "VendorUrl": "https://github.com/riptideio/pymodbus/",
            "ProductName": "Pymodbus Server",
            "ModelName": "Pymodbus Server",
        }
    )
    return args

    class RuntimeManager:
    """Own live devices and their tasks; commits never yield to protocol access."""

    def __init__(self, allow_scripts: bool = False) -> None:
        self.allow_scripts = allow_scripts
        self.deployment: Deployment | None = None
        self.context = RuntimeContext([])
        self.instances: dict[str, Instance] = {}
        self.devices: dict[str, Any] = {}
        self.tasks: dict[str, asyncio.Task] = {}
        self.errors: dict[str, str] = {}
        self.lock = asyncio.Lock()

    def plan(self, desired: Deployment) -> dict:
        if self.deployment:
            if desired.path != self.deployment.path:
                raise ValueError("deployment name belongs to a different file")
            if desired.name != self.deployment.name:
                raise ValueError("deployment name cannot change")
            if (desired.bind, desired.port) != (
                self.deployment.bind,
                self.deployment.port,
            ):
                raise ValueError("listener changed; run down before up")
        added, replaced, kept = [], [], []
        for key, item in desired.instances.items():
            old = self.instances.get(key)
            if old is None:
                added.append(key)
            elif item.td["id"] != old.td["id"]:
                raise ValueError(
                    f"{key}: TD id changed; remove the device before recreating"
                )
            elif item.fingerprint != old.fingerprint:
                replaced.append(key)
            else:
                kept.append(key)
        return {
            "added": added,
            "replaced": replaced,
            "removed": sorted(self.instances.keys() - desired.instances.keys()),
            "kept": kept,
        }

    async def apply(self, desired: Deployment, dry_run: bool = False) -> dict:
        plan = self.plan(desired)
        # Only changed candidates initialize; repeated up cannot rerun live hooks.
        # Building validates schemas and executes behavior scripts; keep it off the
        # event loop so Modbus clients of the running devices are still served.
        candidates = {
            key: await asyncio.to_thread(
                desired.instances[key].build, self.allow_scripts, preview=dry_run
            )
            for key in plan["added"] + plan["replaced"]
        }
        if dry_run:
            return plan
        retired = [self.tasks[key] for key in plan["removed"] + plan["replaced"]]
        next_devices = {
            key: candidates[key] if key in candidates else self.devices[key]
            for key in desired.instances
        }
        next_tasks = {key: self.tasks[key] for key in plan["kept"]}
        for key, device in candidates.items():
            next_tasks[key] = asyncio.create_task(self.run_device(key, device))
        # The supported datastore operations and hooks do not suspend. Swapping
        # this registry in the same event loop is also atomic across write echoes.
        self.context.devices = {
            device.unit_id: device
            for key, device in next_devices.items()
            if key not in self.errors or key in candidates
        }
        for task in retired:
            task.cancel()
        self.instances = desired.instances
        self.devices = next_devices
        self.tasks = next_tasks
        self.errors = {
            key: self.errors[key] for key in plan["kept"] if key in self.errors
        }
        self.deployment = desired
        await asyncio.gather(*retired, return_exceptions=True)
        LOG.info("Deployment applied: %s", plan)
        return plan

    async def run_device(self, key: str, device: Any) -> None:
        try:
            await device.run()
        except Exception as exc:
            self.errors[key] = str(exc)
            # An invalid behavior must not keep serving apparently healthy data.
            self.context.devices.pop(device.unit_id, None)
            LOG.exception("Device %s failed; other devices remain online", key)

    async def reset(self, key: str) -> dict:
        if key not in self.instances:
            raise ValueError(f"unknown device: {key}")
        candidate = self.instances[key].build(self.allow_scripts)
        old = self.tasks[key]
        self.context.devices[candidate.unit_id] = candidate
        self.devices[key] = candidate
        self.tasks[key] = asyncio.create_task(self.run_device(key, candidate))
        self.errors.pop(key, None)
        old.cancel()
        await asyncio.gather(old, return_exceptions=True)
        return {"reset": key}

    def status(self) -> dict:
        return {
            "running": True,
            "name": self.deployment.name if self.deployment else None,
            "source": str(self.deployment.path) if self.deployment else None,
            "pid": os.getpid(),
            "devices": [
                {
                    "id": key,
                    "tdId": self.instances[key].td["id"],
                    "unitId": device.unit_id,
                    "state": "failed" if key in self.errors else "running",
                    "error": self.errors.get(key),
                }
                for key, device in self.devices.items()
            ],
        }

    async def close(self) -> None:
        self.context.devices = {}
        for task in self.tasks.values():
            task.cancel()
        await asyncio.gather(*self.tasks.values(), return_exceptions=True)

    def paths(name: str) -> tuple[Path, Path, Path]:
    """Use a per-user private directory and a short, name-derived socket path."""
    if os.name != "posix":
        raise ValueError("managed commands currently require POSIX (Linux/macOS/WSL)")
    root = Path(tempfile.gettempdir()) / f"modsim-{os.getuid()}"
    root.mkdir(mode=0o700, exist_ok=True)
    info = root.lstat()
    if not stat.S_ISDIR(info.st_mode) or info.st_uid != os.getuid():
        raise ValueError("unsafe runtime directory")
    if stat.S_IMODE(info.st_mode) != 0o700:
        raise ValueError("runtime directory must have permissions 0700")
    key = hashlib.sha256(name.encode()).hexdigest()[:24]
    return root / f"{key}.sock", root / f"{key}.lock", root / f"{key}.log"

    async def serve(
    path: Path, allow_scripts: bool = False, *, detached: bool = False
) -> None:
    """Hold the deployment lock through listener and task shutdown."""
    import fcntl

    desired = await asyncio.to_thread(load_deployment, path)
    socket_path, lock_path, log_path = paths(desired.name)
    with lock_path.open("a") as lease:
        try:
            fcntl.flock(lease, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError as exc:
            raise ValueError(
                "deployment is already starting or running; retry up"
            ) from exc
        manager = RuntimeManager(allow_scripts)
        stop = asyncio.Event()
        server: Any = ModbusTcpServer(
            context=SimDevice(
                id=0,
                simdata=[SimData(0, datatype=DataType.INVALID)],
            ),
            address=(desired.bind, desired.port),
        )
        server.context = manager.context
        control = None
        handlers: set[asyncio.Task] = set()
        loop = asyncio.get_running_loop()
        signals = []
        file_handler = None
        if not detached:
            file_handler = logging.FileHandler(log_path)
            os.chmod(log_path, 0o600)
            logging.getLogger().addHandler(file_handler)

        async def handle(
            reader: asyncio.StreamReader, writer: asyncio.StreamWriter
        ) -> None:
            task = asyncio.current_task()
            if task is not None:
                handlers.add(task)
            try:
                # Read before locking so an idle connection cannot stall other commands.
                payload = await asyncio.wait_for(reader.readline(), 5)
                message = json.loads(payload)
                if message.get("source") != str(desired.path):
                    raise ValueError("deployment name belongs to a different file")
                action = message.get("action")
                async with manager.lock:
                    if action == "up":
                        if message.get("allow_scripts") and not manager.allow_scripts:
                            raise ValueError(
                                "script permission requires down, then up --allow-scripts"
                            )
                        candidate = await asyncio.to_thread(
                            load_deployment, desired.path
                        )
                        result = await manager.apply(
                            candidate, bool(message.get("dry_run"))
                        )
                    elif action == "ps":
                        result = manager.status()
                    elif action == "reset":
                        result = await manager.reset(message.get("device", ""))
                    elif action == "export":
                        result = {
                            "tds": {
                                key: item.td for key, item in manager.instances.items()
                            }
                        }
                    elif action == "down":
                        result = {"stopping": True}
                        stop.set()
                    else:
                        raise ValueError("unknown management action")
                response = result
            except Exception as exc:
                response = {"error": str(exc)}
            try:
                writer.write(json.dumps(response).encode() + b"\n")
                await writer.drain()
            finally:
                writer.close()
                with suppress(ConnectionError):
                    await writer.wait_closed()
                if task is not None:
                    handlers.discard(task)

        try:
            await manager.apply(desired)
            if not await server.listen():
                raise OSError(f"cannot listen on {desired.bind}:{desired.port}")
            # The lease proves that a leftover socket cannot belong to a live owner.
            socket_path.unlink(missing_ok=True)
            control = await asyncio.start_unix_server(
                handle, path=str(socket_path), limit=MAX_REQUEST
            )
            os.chmod(socket_path, 0o600)
            for signum in (signal.SIGINT, signal.SIGTERM):
                loop.add_signal_handler(signum, stop.set)
                signals.append(signum)
            LOG.info(
                "Deployment %s ready at %s:%s", desired.name, desired.bind, desired.port
            )
            await stop.wait()
        finally:
            if control:
                control.close()
                await control.wait_closed()
            for task in list(handlers):
                task.cancel()
            await asyncio.gather(*handlers, return_exceptions=True)
            for signum in signals:
                loop.remove_signal_handler(signum)
            await manager.close()
            await server.shutdown()
            socket_path.unlink(missing_ok=True)
            if file_handler:
                logging.getLogger().removeHandler(file_handler)
                file_handler.close()

    class DeploymentLoader(yaml.SafeLoader):  # type: ignore[no-any-unimported]
    """Reject aliases and duplicate keys rather than expanding ambiguous YAML."""

    def compose_node(self, parent: Any, index: Any) -> Any:
        if self.check_event(yaml.AliasEvent):
            raise ValueError("YAML aliases are unsupported")
        depth = getattr(self, "deployment_depth", 0)
        if depth >= 64:
            raise ValueError("deployment nesting exceeds 64 containers")
        self.deployment_depth = depth + 1
        try:
            return super().compose_node(parent, index)
        finally:
            self.deployment_depth = depth

    def construct_mapping(self, node: Any, deep: bool = False) -> dict:
        result = {}
        for key_node, value_node in node.value:
            key = self.construct_object(key_node, deep=deep)
            if not isinstance(key, str):
                raise ValueError("YAML object keys must be strings")
            if key in result:
                raise ValueError(f"duplicate YAML key: {key}")
            result[key] = self.construct_object(value_node, deep=deep)
        return result

    def deployment_root(path: Path) -> Path:
    """Use the repository root for checked-in examples, config directory otherwise."""
    for parent in path.parent.parents:
        marker = parent / ".git"
        if marker.is_file() or (marker.is_dir() and any(marker.iterdir())):
            return parent
    marker = path.parent / ".git"
    if marker.is_file() or (marker.is_dir() and any(marker.iterdir())):
        return path.parent
    return path.parent

    def load_deployment(path: Path, *, include_data: bool = True) -> Deployment:
    """Resolve and validate every instance before the runtime can be changed."""
    path = path.resolve()
    if path.suffix.lower() not in (".yaml", ".yml"):
        raise ValueError("deployment file must use .yaml or .yml")
    spec = read_yaml(path)
    validate(spec, MANIFEST)
    name = spec.get("name", path.parent.name)
    validate({"name": name}, obj({"name": IDENTIFIER}, ("name",)))
    bind = spec.get("server", {}).get("bind", "127.0.0.1")
    port = spec.get("server", {}).get("port", 5020)
    root = deployment_root(path)
    instances: dict[str, Instance] = {}
    units: set[int] = set()
    identities: set[str] = set()
    for item in spec["devices"]:
        key = item["id"]
        if key in instances:
            raise ValueError(f"duplicate device id: {key}")
        source = item["model"]
        if not source.startswith("https://"):
            source = local_path(root, source)
        document = model_document(source)
        if not isinstance(document, dict):
            raise ValueError(f"{key}: model must be an object")
        types = document.get("@type", [])
        if isinstance(types, str):
            types = [types]
        if "tm:ThingModel" not in types:
            raise ValueError(f"{key}: deployment v2 requires a TM; use --config for TD")
        parameters = dict(item.get("parameters", {}))
        if "AUTHORITY" in parameters:
            raise ValueError(
                "use HOST and PORT; AUTHORITY is derived by the interpreter"
            )
        # A TM need not declare a Thing ID. An explicit instance binding may
        # provide it without changing the reusable model or inventing identity.
        if "THING_ID" in parameters and "id" not in document:
            document = copy.deepcopy(document)
            document["id"] = "{{THING_ID}}"
        host = parameters.pop("HOST", None)
        if host is None:
            if bind in ("0.0.0.0", "::", "*"):
                raise ValueError(f"{key}: wildcard listener requires parameters.HOST")
            host = bind
        instance_port = parameters.pop("PORT", port)
        unit = parameters.pop("UNIT_ID", None)
        if not isinstance(host, str):
            raise ValueError(f"{key}: HOST must be a string")
        td = export_td(
            document, host=host, port=instance_port, unit_id=unit, parameters=parameters
        )
        if len(canonical(td)) > MAX_BYTES:
            raise ValueError(f"{key}: generated TD exceeds 1 MiB")
        model = parse_model(td)
        identity = td.get("id")
        if not isinstance(identity, str) or not identity:
            raise ValueError(f"{key}: TM must instantiate a stable TD id")
        if identity in identities or model.unit_id in units:
            raise ValueError(f"{key}: duplicate TD id or Unit ID")
        units.add(model.unit_id)
        identities.add(identity)
        data: dict = {}
        script_path, script_source = None, None
        if include_data and item.get("data"):
            data = load_simulation_data(
                read_yaml(Path(local_path(root, item["data"]))),
                title=model.title,
                properties=json.loads(model.properties_json).keys(),
            ).data
        if include_data and item.get("script"):
            script_path = local_path(root, item["script"])
            # Scripts execute with the simulator's rights; keep them with the deployment.
            if not Path(script_path).is_relative_to(root.resolve()):
                raise ValueError(f"{key}: script must stay inside {root}")
            with Path(script_path).open("rb") as stream:
                script_source = stream.read(MAX_BYTES + 1)
            if len(script_source) > MAX_BYTES:
                raise ValueError("script exceeds 1 MiB")
        script_parameters = dict(item.get("scriptParameters", {}))
        fingerprint = hashlib.sha256(
            canonical([item, document, td, data, script_path]) + (script_source or b"")
        ).hexdigest()
        instances[key] = Instance(
            key,
            td,
            model,
            fingerprint,
            data,
            script_path,
            script_source,
            script_parameters,
        )
    return Deployment(path, name, bind, port, instances)

    def read_yaml(path: Path) -> dict:
    """Read JSON or safe YAML with bounded size, depth and JSON-compatible values."""
    if path.suffix.lower() == ".json":
        value = load_document(path)
    else:
        with path.open("rb") as stream:
            payload = stream.read(MAX_BYTES + 1)
        if len(payload) > MAX_BYTES:
            raise ValueError("deployment exceeds 1 MiB")
        try:
            value = yaml.load(payload, Loader=DeploymentLoader)
        except yaml.YAMLError as exc:
            raise ValueError(f"{path}: invalid YAML: {exc}") from exc
    if not isinstance(value, dict):
        raise ValueError(f"{path}: expected an object")
    try:
        canonical(value)
    except (TypeError, ValueError, RecursionError) as exc:
        raise ValueError(f"{path}: expected finite JSON-compatible values") from exc
    return value
